-- ==============================================================================
-- 2. TABLE CREATION
-- ==============================================================================

-- Master Products Catalog
CREATE TABLE products (
    id BIGSERIAL PRIMARY KEY,
    sku VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(100) NOT NULL,
    base_price NUMERIC(10,2) NOT NULL,
    default_warranty_months INT DEFAULT 12,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- Advanced Stock Tracking (By Serial Number)
CREATE TABLE inventory_serials (
    id BIGSERIAL PRIMARY KEY,
    product_id BIGINT REFERENCES products(id) ON DELETE CASCADE,
    serial_number VARCHAR(100) UNIQUE NOT NULL,
    status VARCHAR(20) DEFAULT 'IN_STOCK', -- IN_STOCK, SOLD, DEFECTIVE, RETURNED, LOST
    warehouse_location VARCHAR(50),
    received_date TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- Audit Trail for Inventory Adjustments (RMA / Defects)
CREATE TABLE inventory_logs (
    id BIGSERIAL PRIMARY KEY,
    inventory_serial_id BIGINT REFERENCES inventory_serials(id) ON DELETE CASCADE,
    previous_status VARCHAR(20),
    new_status VARCHAR(20),
    changed_by BIGINT, -- Corresponds to user_id in your users table
    changed_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    notes TEXT
);

-- Discount Rules Engine
CREATE TABLE discounts (
    id BIGSERIAL PRIMARY KEY,
    code VARCHAR(50) UNIQUE NOT NULL,
    discount_type VARCHAR(20) NOT NULL, -- PERCENTAGE or FIXED
    discount_value NUMERIC(10,2) NOT NULL,
    is_active BOOLEAN DEFAULT TRUE
);

-- POS Invoices (Header)
CREATE TABLE pos_invoices (
    id BIGSERIAL PRIMARY KEY,
    receipt_no VARCHAR(50) UNIQUE NOT NULL,
    subtotal NUMERIC(10,2) NOT NULL,
    total_discount NUMERIC(10,2) DEFAULT 0.00,
    grand_total NUMERIC(10,2) NOT NULL,
    payment_method VARCHAR(20) NOT NULL, -- CASH, KHQR, CARD
    cashier_id BIGINT NOT NULL,
    sale_date TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- POS Invoice Items (Line details with specific Serial & Warranty)
CREATE TABLE pos_invoice_items (
    id BIGSERIAL PRIMARY KEY,
    invoice_id BIGINT REFERENCES pos_invoices(id) ON DELETE CASCADE,
    product_id BIGINT REFERENCES products(id),
    inventory_serial_id BIGINT REFERENCES inventory_serials(id),
    original_price NUMERIC(10,2) NOT NULL,
    discount_id BIGINT REFERENCES discounts(id),
    discount_amount NUMERIC(10,2) DEFAULT 0.00,
    final_price NUMERIC(10,2) NOT NULL,
    warranty_start_date TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    warranty_end_date TIMESTAMPTZ NOT NULL,
    warranty_status VARCHAR(20) DEFAULT 'ACTIVE' -- ACTIVE, EXPIRED, VOID
);

-- ==============================================================================

-- VIEW: Current Available Stock
CREATE VIEW view_current_stock AS
SELECT 
    p.sku,
    p.name AS product_name,
    COUNT(s.id) AS available_quantity
FROM products p
LEFT JOIN inventory_serials s ON p.id = s.product_id AND s.status = 'IN_STOCK'
GROUP BY p.sku, p.name;

-- VIEW: Daily Stock In (Receiving)
CREATE VIEW view_stock_in_report AS
SELECT 
    DATE(s.received_date) AS date_received,
    p.sku,
    p.name AS product_name,
    COUNT(s.id) AS total_qty_in
FROM inventory_serials s
JOIN products p ON s.product_id = p.id
GROUP BY DATE(s.received_date), p.sku, p.name;

-- VIEW: Daily Stock Out (Sales)
CREATE VIEW view_stock_out_report AS
SELECT 
    DATE(inv.sale_date) AS date_sold,
    p.sku,
    p.name AS product_name,
    COUNT(item.id) AS total_qty_out,
    SUM(item.final_price) AS total_sales_revenue
FROM pos_invoices inv
JOIN pos_invoice_items item ON inv.id = item.invoice_id
JOIN products p ON item.product_id = p.id
GROUP BY DATE(inv.sale_date), p.sku, p.name;

លទ្ធផលនៃ Query នេះនឹងបង្រួមទិន្នន័យ ដោយបង្ហាញសរុបតែម្តងថា ក្នុងខែនេះ Soundbox ម៉ូដែលនីមួយៗលក់បានប៉ុន្មានគ្រឿង ចំណាយដើមសរុបអស់ប៉ុន្មាន ចំណូលបានប៉ុន្មាន និងចំណេញសុទ្ធសរុបប៉ុន្មាន។





SELECT 
    TO_CHAR(inv.sale_date, 'YYYY-MM') AS sale_month,
    p.sku,
    p.name AS product_name,
    COUNT(item.id) AS total_units_sold,
    SUM(s.purchase_price) AS total_cost,
    SUM(item.final_price) AS total_revenue,
    SUM(item.final_price - s.purchase_price) AS total_net_profit
FROM pos_invoice_items item
JOIN inventory_serials s ON item.inventory_serial_id = s.id
JOIN products p ON item.product_id = p.id
JOIN pos_invoices inv ON item.invoice_id = inv.id
-- ដាក់លក្ខខណ្ឌដើម្បីទាញយកតែទិន្នន័យក្នុងខែបច្ចុប្បន្ន (ខែកញ្ញា ឆ្នាំ២០២៦)
WHERE inv.sale_date >= DATE_TRUNC('month', CURRENT_DATE) 
GROUP BY TO_CHAR(inv.sale_date, 'YYYY-MM'), p.sku, p.name
ORDER BY total_net_profit DESC;

ការបន្ថែមទិន្នន័យកែតម្រូវស្តុក (Adjustments ដូចជាទំនិញខូច ឬបាត់បង់) ចូលទៅក្នុងរបាយការណ៍ប្រាក់ចំណេញ គឺជារឿងចាំបាច់បំផុត ដើម្បីរកឱ្យឃើញនូវ ប្រាក់ចំណេញសុទ្ធពិតប្រាកដ (True Net Profit)។




WITH SalesSummary AS (
    -- ផ្នែកទី១៖ គណនាចំណូល និងប្រាក់ចំណេញពីការលក់ (Sales)
    SELECT 
        TO_CHAR(inv.sale_date, 'YYYY-MM') AS report_month,
        COUNT(item.id) AS qty_sold,
        SUM(s.purchase_price) AS cost_of_goods_sold, -- ដើមទុនទំនិញលក់ចេញ
        SUM(item.final_price) AS total_revenue -- ចំណូលសរុប
    FROM pos_invoice_items item
    JOIN inventory_serials s ON item.inventory_serial_id = s.id
    JOIN pos_invoices inv ON item.invoice_id = inv.id
    GROUP BY TO_CHAR(inv.sale_date, 'YYYY-MM')
),
AdjustmentSummary AS (
    -- ផ្នែកទី២៖ គណនាការខាតបង់ពីការកែតម្រូវស្តុក (Defective/Lost) ផ្អែកលើ Logs
    SELECT 
        TO_CHAR(l.changed_at, 'YYYY-MM') AS report_month,
        COUNT(l.id) AS qty_adjusted,
        SUM(s.purchase_price) AS total_adjustment_loss -- ដើមទុនដែលខាតបង់
    FROM inventory_logs l
    JOIN inventory_serials s ON l.inventory_serial_id = s.id
    WHERE l.new_status IN ('DEFECTIVE', 'LOST')
    GROUP BY TO_CHAR(l.changed_at, 'YYYY-MM')
)
-- ផ្នែកទី៣៖ ភ្ជាប់ទិន្នន័យទាំង២ បង្ហាញជារបាយការណ៍ចុងក្រោយ
SELECT 
    COALESCE(s.report_month, a.report_month) AS month,
    COALESCE(s.qty_sold, 0) AS items_sold,
    COALESCE(s.total_revenue, 0.00) AS total_revenue,
    
    COALESCE(a.qty_adjusted, 0) AS items_defective_lost,
    COALESCE(a.total_adjustment_loss, 0.00) AS adjustment_loss,
    
    -- គណនាប្រាក់ចំណេញសុទ្ធចុងក្រោយ៖ ចំណូល - ដើមទុនលក់ចេញ - ដើមទុនខាតបង់
    (COALESCE(s.total_revenue, 0.00) - COALESCE(s.cost_of_goods_sold, 0.00) - COALESCE(a.total_adjustment_loss, 0.00)) AS final_net_profit
    
FROM SalesSummary s
FULL OUTER JOIN AdjustmentSummary a ON s.report_month = a.report_month
ORDER BY month DESC;

របៀបឆែកមើលថាតើទំនិញ (Serial Number ណាមួយ) នៅសល់ធានាឬអត់


SELECT 
    s.serial_number,
    s.status AS inventory_status,
    p.name AS product_name,
    COALESCE(i.receipt_no, 'គ្មានវិក្កយបត្រ (មិនទាន់លក់)') AS receipt_no,
    pii.warranty_start_date,
    pii.warranty_end_date,
    CASE 
        WHEN s.status = 'IN_STOCK' THEN 'ទំនិញថ្មីក្នុងស្តុក (មិនទាន់លក់)'
        WHEN s.status = 'DEFECTIVE' THEN 'ទំនិញខូច (Defective)'
        WHEN s.status = 'SOLD' AND pii.warranty_end_date >= CURRENT_DATE THEN 'ស្ថិតក្នុងរយៈពេលធានា (Under Warranty)'
        WHEN s.status = 'SOLD' AND pii.warranty_end_date < CURRENT_DATE THEN 'អស់សុពលភាពធានា (Expired)'
        ELSE 'មិនមានទិន្នន័យធានា'
    END AS evaluation_result
FROM inventory_serials s
JOIN products p ON s.product_id = p.id
LEFT JOIN pos_invoice_items pii ON s.id = pii.inventory_serial_id
LEFT JOIN pos_invoices i ON pii.invoice_id = i.id
WHERE s.serial_number = 'SN-PRO-006'; -- ប្តូរជា Serial Number ជាក់ស្តែងដែលចង់ឆែក
